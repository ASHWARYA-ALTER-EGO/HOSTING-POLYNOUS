// ─── API helpers ──────────────────────────────────────────────────────────────

async function apiNewResearch() {
  const res = await fetch("/api/research/new", { method: "POST" });
  return res.json();
}

async function apiBeginDebate(topic) {
  const res = await fetch("/api/debate/begin", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ topic }),
  });
  return res.json();
}

async function apiSelectTopic(topic) {
  const res = await fetch("/api/debate/select-topic", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ topic }),
  });
  return res.json();
}

async function apiNavigate(page) {
  const res = await fetch("/api/nav", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ page }),
  });
  return res.json();
}

async function apiDisconnect() {
  const res = await fetch("/api/auth/disconnect", { method: "POST" });
  return res.json();
}

async function apiToggleSidebar(collapsed) {
  const res = await fetch("/api/ui/sidebar", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ collapsed }),
  });
  return res.json();
}

// ─── Globe (loads Three.js dynamically if not present) ───────────────────────

function Globe({ containerRef }) {
  useEffect(() => {
    let animId;
    let renderer;

    function initGlobe(THREE) {
      const container = containerRef.current;
      if (!container) return;

      const width = container.clientWidth;
      const height = container.clientHeight;

      const scene = new THREE.Scene();
      const camera = new THREE.PerspectiveCamera(45, width / height, 0.1, 1000);
      camera.position.z = 2.4;

      renderer = new THREE.WebGLRenderer({ alpha: true, antialias: true });
      renderer.setSize(width, height);
      renderer.setPixelRatio(window.devicePixelRatio);
      // Insert canvas as first child (before the overlay div)
      container.insertBefore(renderer.domElement, container.firstChild);

      const brandRed = new THREE.Color(0xff0022);

      // Core sphere
      const sphereGeom = new THREE.SphereGeometry(1, 64, 64);
      const sphereMat = new THREE.MeshPhongMaterial({
        color: 0x050505,
        transparent: true,
        opacity: 0.95,
        shininess: 30,
      });
      const core = new THREE.Mesh(sphereGeom, sphereMat);
      core.rotation.y = -Math.PI / 4;
      core.rotation.x = Math.PI / 8;
      scene.add(core);

      // Continent shader via texture
      const textureLoader = new THREE.TextureLoader();
      textureLoader.crossOrigin = "anonymous";
      textureLoader.load(
        "https://raw.githubusercontent.com/mrdoob/three.js/master/examples/textures/planets/earth_specular_2048.jpg",
        (texture) => {
          const customMaterial = new THREE.ShaderMaterial({
            uniforms: {
              tEarth: { value: texture },
              color: { value: brandRed },
            },
            vertexShader: `
              varying vec2 vUv;
              void main() {
                vUv = uv;
                gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
              }
            `,
            fragmentShader: `
              uniform sampler2D tEarth;
              uniform vec3 color;
              varying vec2 vUv;
              void main() {
                vec4 texColor = texture2D(tEarth, vUv);
                float continentMask = 1.0 - texColor.r;
                float dotsX = sin(vUv.x * 1200.0);
                float dotsY = sin(vUv.y * 1200.0);
                float dots = (dotsX * dotsY) * 0.5 + 0.5;
                float edge = smoothstep(0.4, 0.6, dots);
                float alpha = continentMask * edge * 0.9;
                if (alpha < 0.1) discard;
                gl_FragColor = vec4(color * 1.5, alpha);
              }
            `,
            transparent: true,
            blending: THREE.AdditiveBlending,
          });
          const earthMap = new THREE.Mesh(
            new THREE.SphereGeometry(1.01, 64, 64),
            customMaterial
          );
          core.add(earthMap);
        }
      );

      // Wireframe
      core.add(
        new THREE.Mesh(
          new THREE.SphereGeometry(1.02, 32, 32),
          new THREE.MeshBasicMaterial({
            color: brandRed,
            wireframe: true,
            transparent: true,
            opacity: 0.15,
          })
        )
      );

      // Points
      core.add(
        new THREE.Points(
          new THREE.SphereGeometry(1.03, 48, 48),
          new THREE.PointsMaterial({
            color: brandRed,
            size: 0.015,
            transparent: true,
            opacity: 0.8,
            blending: THREE.AdditiveBlending,
          })
        )
      );

      // Atmosphere glow
      const atmosphere = new THREE.Mesh(
        new THREE.SphereGeometry(1.15, 64, 64),
        new THREE.ShaderMaterial({
          vertexShader: `
            varying vec3 vNormal;
            void main() {
              vNormal = normalize(normalMatrix * normal);
              gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
            }
          `,
          fragmentShader: `
            varying vec3 vNormal;
            void main() {
              float intensity = pow(0.65 - dot(vNormal, vec3(0,0,1.0)), 4.0);
              gl_FragColor = vec4(1.0, 0.0, 0.13, 1.0) * intensity * 2.0;
            }
          `,
          side: THREE.BackSide,
          blending: THREE.AdditiveBlending,
          transparent: true,
          depthWrite: false,
        })
      );
      scene.add(atmosphere);

      // Decorative rings
      const createRing = (radius, rotX, rotY, opacity) => {
        const ring = new THREE.Mesh(
          new THREE.RingGeometry(radius, radius + 0.005, 64),
          new THREE.MeshBasicMaterial({
            color: brandRed,
            side: THREE.DoubleSide,
            transparent: true,
            opacity,
            blending: THREE.AdditiveBlending,
          })
        );
        ring.rotation.x = rotX;
        ring.rotation.y = rotY;
        scene.add(ring);
        return ring;
      };
      const ring1 = createRing(1.4, Math.PI / 2.5, 0, 0.1);
      const ring2 = createRing(1.6, Math.PI / 1.5, Math.PI / 8, 0.05);

      // Lights
      scene.add(new THREE.AmbientLight(0xffffff, 0.2));
      const pl = new THREE.PointLight(brandRed, 3, 10);
      pl.position.set(5, 5, 5);
      scene.add(pl);
      const bl = new THREE.PointLight(brandRed, 2, 10);
      bl.position.set(-5, 3, -5);
      scene.add(bl);

      // Animation
      const animate = () => {
        animId = requestAnimationFrame(animate);
        const time = Date.now() * 0.0005;
        core.rotation.y += 0.001;
        ring1.rotation.z -= 0.002;
        ring2.rotation.z += 0.001;
        const pulse = 1.0 + Math.sin(time * 2.0) * 0.01;
        atmosphere.scale.set(pulse, pulse, pulse);
        renderer.render(scene, camera);
      };
      animate();

      // Resize handler
      const onResize = () => {
        if (!container) return;
        camera.aspect = container.clientWidth / container.clientHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(container.clientWidth, container.clientHeight);
      };
      window.addEventListener("resize", onResize);

      // Store cleanup reference
      container._globeCleanup = () => {
        cancelAnimationFrame(animId);
        window.removeEventListener("resize", onResize);
        renderer.dispose();
        if (renderer.domElement.parentNode === container) {
          container.removeChild(renderer.domElement);
        }
      };
    }

    function loadThreeAndInit() {
      if (window.THREE) {
        initGlobe(window.THREE);
        return;
      }
      const script = document.createElement("script");
      script.src = "https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js";
      script.onload = () => initGlobe(window.THREE);
      document.head.appendChild(script);
    }

    loadThreeAndInit();

    return () => {
      const container = containerRef.current;
      if (container && container._globeCleanup) {
        container._globeCleanup();
        delete container._globeCleanup;
      }
    };
  }, [containerRef]);

  return null;
}
